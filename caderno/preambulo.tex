% ---- fontes: Libertinus + reserva para símbolos que ela não tem (●, ✔, ⚠...) ----
% Funciona no Windows (Segoe UI Symbol, Consolas) e no Linux/TeX Live (DejaVu).
\IfFontExistsTF{Segoe UI Symbol}{%
  \directlua{luaotfload.add_fallback("simbolos", {"Segoe UI Symbol:mode=harf;", "Cambria Math:mode=harf;"})}%
}{%
  \directlua{luaotfload.add_fallback("simbolos", {"DejaVu Sans:mode=harf;", "DejaVu Sans Mono:mode=harf;"})}%
}
\setmainfont{Libertinus Serif}[RawFeature={fallback=simbolos}]
\setsansfont{Libertinus Sans}[RawFeature={fallback=simbolos}]
\IfFontExistsTF{Consolas}{%
  \setmonofont{Consolas}[Scale=0.84, RawFeature={fallback=simbolos}]%
}{%
  \setmonofont{DejaVu Sans Mono}[Scale=0.80, RawFeature={fallback=simbolos}]%
}
\setmathfont{Libertinus Math}
